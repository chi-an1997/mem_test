`include "uvm_macros.svh"
class mem_env extends uvm_env;
    mem_agent mem_ctr_agent; 
    mem_sb sb;
    `uvm_component_utils (mem_env)

    function new (string name="mem_env", uvm_component parent); 
        super.new(name, parent);
    endfunction

    virtual function void build_phase (uvm_phase phase); 
        super.build_phase (phase);
        mem_ctr_agent = mem_agent::type_id::create("mem_ctr_agent", this);

        $display ("env start test case............................");
        sb = mem_sb::type_id::create("sb", this); 
    endfunction

    virtual function void connect_phase (uvm_phase phase); 
        super.connect_phase (phase);
        mem_ctr_agent.drv.drv_port.connect(sb.exp_port); 
        mem_ctr_agent.mnt.ap.connect(sb.act_port);
    endfunction
endclass
